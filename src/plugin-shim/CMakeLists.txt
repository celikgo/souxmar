# souxmar_plugin_shim — the Windows half of plugin symbol resolution.
#
# A souxmar plugin leaves the host ABI undefined and lets the loader bind it
# to whichever process dlopen()s it. ELF allows that; Mach-O opts in via
# `-undefined dynamic_lookup` (see cmake/SouxmarPlugin.cmake); Windows does
# not allow it at all — every symbol a DLL references must be bound at link
# time to a named module.
#
# Binding to a host's import library is not usable here. souxmar has several
# hosts — the CLI, both test runners, the conformance tool, the desktop
# bridge — and a plugin has to load into any of them. A DLL that imports
# from `souxmar.exe` cannot be loaded by `souxmar_unit_tests.exe`.
#
# So on Windows the plugin links this static shim instead. Each ABI function
# becomes a thunk that resolves itself on first call through
# GetProcAddress(GetModuleHandle(NULL), ...) — the running executable, which
# is exactly the rule the other two platforms apply for free. The hosts
# already export the symbols: CMAKE_EXECUTABLE_ENABLE_EXPORTS in
# cmake/SouxmarOptions.cmake, whose comment calls this out as the missing
# Windows half.
#
# The ABI itself is untouched. Same symbol names, same signatures, same
# semantics — include/souxmar-c/** does not change, and neither does the
# plugin source. This is a link-time detail of one platform.
#
# shim_win32.c is generated by scripts/gen-windows-plugin-shim.py and
# committed so the thunks are reviewable; `ci.yml` re-runs the generator with
# --check-only so a header change cannot leave it stale.
#
# Not linked into hosts, ever: the host defines these symbols for real, and
# the shim would shadow them with thunks that resolve back to the host.

if(NOT WIN32)
  return()
endif()

add_library(souxmar_plugin_shim STATIC shim_win32.c)
add_library(souxmar::plugin_shim ALIAS souxmar_plugin_shim)

target_link_libraries(souxmar_plugin_shim
  PUBLIC
    souxmar::public_headers
  PRIVATE
    souxmar::sanitizers
)

# Deliberately not souxmar::warnings. This is generated code compiled as C,
# and the project's warning set is tuned for hand-written C++ — a conversion
# warning in a thunk that forwards its arguments verbatim would be noise with
# nothing to fix.
set_target_properties(souxmar_plugin_shim PROPERTIES
  POSITION_INDEPENDENT_CODE ON
  FOLDER "plugins"
)
